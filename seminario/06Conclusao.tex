\begin{frame}

    \frametitle{Conclusão}

    \begin{itemize}
        \item BCBF = \textbf{CBF aplicada à crença} do EKF, não ao estado.
        \item Segurança passa a ser um limite na \textbf{probabilidade de violação}.
        \item Garantias mesmo com medições discretas e outliers (margem $\gamma$).
        \item Continua sendo um QP: \textbf{tempo real} e aplicável a qualquer robô com EKF.
    \end{itemize}

    \vspace{3mm}

    \textbf{Limitações e trabalhos futuros}
    \begin{itemize}
        \item Supõe crença Gaussiana; futuro: filtros de partículas.
        \item Reativo, sujeito a mínimos locais; futuro: combinar com MPC/planejadores.
    \end{itemize}

\end{frame}
